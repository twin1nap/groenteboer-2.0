                               $"VALUES (@receiptId{i}, @productId{i}, @amount{i}, @price{i}, @priceType{i});";

            cmd.Parameters.AddWithValue($"@receiptId{i}", receiptId);
            cmd.Parameters.AddWithValue($"@productId{i}", products[i].Id);
            cmd.Parameters.AddWithValue($"@amount{i}", products[i].Amount);
            cmd.Parameters.AddWithValue($"@price{i}", products[i].Price);
